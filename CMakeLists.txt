cmake_minimum_required(VERSION 3.21)
project(DevLink VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

option(DEVLINK_BUILD_BACKEND "Build backend" ON)
option(DEVLINK_BUILD_DESKTOP "Build desktop client" ON)

if(DEVLINK_BUILD_BACKEND OR DEVLINK_BUILD_DESKTOP)
    find_package(Qt6 6.6 REQUIRED COMPONENTS Core Network WebSockets)
endif()
if(DEVLINK_BUILD_BACKEND)
    find_package(Qt6 6.6 REQUIRED COMPONENTS HttpServer)
    add_subdirectory(backend)
endif()
if(DEVLINK_BUILD_DESKTOP)
    find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets)
    add_subdirectory(desktop)
endif()
